{{- /* Returns data/projects.json repos with live description, language, stars, and forks from the GitHub API. Repos keep their stored values when a request fails. GITHUB_TOKEN raises the rate limit when it is set. */ -}} {{- $owner := hugo.Data.projects.owner | default "mozanunal" -}} {{- $headers := dict "Accept" "application/vnd.github+json" "X-GitHub-Api-Version" "2022-11-28" -}} {{- with getenv "GITHUB_TOKEN" -}} {{- $headers = merge $headers (dict "Authorization" (printf "Bearer %s" .)) -}} {{- end -}} {{- $out := slice -}} {{- range $repo := hugo.Data.projects.repos -}} {{- $url := printf "https://api.github.com/repos/%s/%s" ($repo.owner | default $owner) $repo.name -}} {{- with try (resources.GetRemote $url (dict "headers" $headers)) -}} {{- with .Err -}} {{- warnf "projects: GitHub request failed for %s, using stored values: %s" $repo.name . -}} {{- else with .Value -}} {{- $live := .Content | transform.Unmarshal -}} {{- $repo = merge $repo (dict "description" ($live.description | default $repo.description) "language" ($live.language | default $repo.language) "stars" $live.stargazers_count "forks" $live.forks_count ) -}} {{- end -}} {{- end -}} {{- $out = $out | append $repo -}} {{- end -}} {{- return $out -}}
